cmake_minimum_required(VERSION 3.24)
project(CellShardEmbeddedConsumer LANGUAGES CXX)
if(NOT CELLSHARD_SOURCE_DIR)
    message(FATAL_ERROR "CELLSHARD_SOURCE_DIR is required")
endif()
set(CELLSHARD_ENABLE_CUDA OFF CACHE BOOL "" FORCE)
set(CELLSHARD_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(CELLSHARD_BUILD_EXPORT OFF CACHE BOOL "" FORCE)
if(NOT CELLSHARD_CELLERATOR_SOURCE_DIR)
    message(FATAL_ERROR "CELLSHARD_CELLERATOR_SOURCE_DIR is required")
endif()
add_subdirectory("${CELLSHARD_SOURCE_DIR}" CellShard)
add_executable(cellshard_embedded_consumer main.cc)
target_link_libraries(cellshard_embedded_consumer PRIVATE CellShard::headers)
target_compile_features(cellshard_embedded_consumer PRIVATE cxx_std_17)
